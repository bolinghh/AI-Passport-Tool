cmake_minimum_required(VERSION 3.16)
project(passport-controller-host C)
enable_testing()
add_executable(test_controller test_controller.c ../main/controller.c ../main/play.c ../main/wire.c)
target_include_directories(test_controller PRIVATE ../main)
if(MSVC)
    target_compile_options(test_controller PRIVATE /W4 /WX /utf-8)
    target_compile_definitions(test_controller PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_options(test_controller PRIVATE -std=c11 -Wall -Wextra -Werror)
endif()
add_test(NAME controller COMMAND test_controller "${CMAKE_CURRENT_SOURCE_DIR}/wire-vectors.txt")
add_executable(test_play test_play.c ../main/play.c ../main/controller.c ../main/wire.c)
target_include_directories(test_play PRIVATE ../main)
if(MSVC)
    target_compile_options(test_play PRIVATE /W4 /WX /utf-8)
else()
    target_compile_options(test_play PRIVATE -std=c11 -Wall -Wextra -Werror)
endif()
add_test(NAME play COMMAND test_play)
add_executable(test_play_store test_play_store.c ../main/play.c)
target_include_directories(test_play_store PRIVATE play_stubs ../main)
if(MSVC)
    target_compile_options(test_play_store PRIVATE /W4 /WX /utf-8)
else()
    target_compile_options(test_play_store PRIVATE -std=c11 -Wall -Wextra -Werror)
endif()
add_test(NAME play_store COMMAND test_play_store)
